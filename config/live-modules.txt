# --- display / GPU (KMS) ---------------------------------------------------
# Without this list, /dev/dri/card0 never appears on VM/VBox/real-PC, and weston
# (desktop session) gets no device -> the screen stays stuck in text mode.
# This kernel has CONFIG_DRM_SIMPLEDRM/SYSFB_SIMPLEFB OFF, so loading some
# native driver is a must; cost ~1 MB (xz) - nothing compared to a desktop.
kernel/drivers/gpu/drm/vmwgfx            # VMware + VirtualBox 'VMSVGA' (VBox default)
kernel/drivers/gpu/drm/vboxvideo         # VirtualBox old 'VBoxVGA' device
kernel/drivers/gpu/drm/qxl               # QEMU/SPICE (-vga qxl)
kernel/drivers/gpu/drm/tiny              # bochs (-vga std) + cirrus (-vga cirrus)
kernel/drivers/gpu/drm/virtio            # virtio-gpu (-vga virtio): fast + clean
kernel/drivers/gpu/drm/ast               # server/BMC on-board graphics
kernel/drivers/gpu/drm/mgag200           # Matrox G200 (in many servers)
kernel/drivers/video/fbdev/vga16fb.ko*   # legacy text-mode fb (/dev/fb0 fallback)
# For physical PCs (bigger, but the real route to a display):
kernel/drivers/gpu/drm/i915              # Intel HD/UHD/Arc-ish (display works without firmware)
kernel/drivers/gpu/drm/nouveau           # old/mid NVIDIA
kernel/drivers/gpu/drm/radeon            # old ATI/AMD (display without firmware)
# amdgpu/xe are deliberately NOT here: they need /lib/firmware/{amdgpu,xe} .bins
# (probe fails otherwise), size ~4 MB+. On these GPUs: text console + 'pk-desktop vnc'
# works; full desktop is better on installed systems (with the firmware pack).
# --- NIC drivers (no udev, so pk-net modprobes them at boot)
